cmake_minimum_required(VERSION 3.20)
project(riscv_emulator VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(riscv_core src/cpu.cpp src/memory.cpp)
target_include_directories(riscv_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(riscv_core PRIVATE /W4)
else()
  target_compile_options(riscv_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(riscv_emulator src/main.cpp)
target_link_libraries(riscv_emulator PRIVATE riscv_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(riscv_tests tests/test_cpu.cpp)
  target_link_libraries(riscv_tests PRIVATE riscv_core)
  add_test(NAME cpu_tests COMMAND riscv_tests)
endif()
